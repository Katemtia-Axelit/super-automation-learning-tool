## 三、执行脚本

### 3.1 预览模式（仅显示将要执行的操作）

```batch
@echo off
setlocal EnableDelayedExpansion
chcp 65001 >nul

echo ============================================
echo   归档清理方案 - 预览模式
echo ============================================
echo.

set "ARCHIVE=_archive"
set "PROJECT_ROOT=%~dp0"

echo [INFO] 项目根目录: %PROJECT_ROOT%
echo [INFO] 归档目录: %ARCHIVE%
echo.

:: 检查 _archive 是否存在
if exist "%ARCHIVE%" (
    echo [WARN] _archive 目录已存在，将在其内创建子目录
) else (
    echo [INFO] 创建 _archive 目录
    mkdir "%ARCHIVE%"
)

echo.
echo ========== 即将移动的文件 ==========
echo.

:: 01_临时脚本
echo [01] 临时脚本 (根目录 _*/ocr_*/extract_*):
dir /b /a-d "%PROJECT_ROOT%_*" 2>nul | findstr /i "_"
dir /b /a-d "%PROJECT_ROOT%ocr_*.py" 2>nul
dir /b /a-d "%PROJECTROOT%ocr_windows.ps1" 2>nul

echo.
echo [02] 缓存数据 (data/extracted/ ai_*/):
dir /b /s /a-d "%PROJECT_ROOT%data\extracted\*" 2>nul
dir /b /s /a-d "%PROJECT_ROOT%data\ai_*\*" 2>nul

echo.
echo [03] 数据库备份:
dir /b /a-d "%PROJECT_ROOT%data\*backup*.db" 2>nul

echo.
echo [04] PPT提取物:
dir /b /s /ad "%PROJECT_ROOT%src\notes\ppt_extracted" 2>nul

echo.
echo [05] 过时文档 (docs/):
dir /b /a-d "%PROJECT_ROOT%docs\*.md" 2>nul | findstr /v "README.md 项目重启方案.md"

echo.
echo [06] 测试脚本 (scripts/):
dir /b /a-d "%PROJECT_ROOT%scripts\p0_*" 2>nul
dir /b /a-d "%PROJECT_ROOT%scripts\p1_*" 2>nul
dir /b /a-d "%PROJECT_ROOT%scripts\debug_*" 2>nul
dir /b /a-d "%PROJECT_ROOT%scripts\fix_*" 2>nul
dir /b /a-d "%PROJECT_ROOT%scripts\check_*" 2>nul

echo.
echo [07] 工具脚本 (tools/):
dir /b /a-d "%PROJECT_ROOT%tools\*" 2>nul

echo.
echo [08] 提示词备份:
if exist "%PROJECT_ROOT%src\notes\prompts\.backups" (
    dir /b /a-d "%PROJECT_ROOT%src\notes\prompts\.backups\*" 2>nul
)

echo.
echo [09] 审计报告:
dir /b /s /a-d "%PROJECT_ROOT%*审计*" 2>nul
dir /b /s /a-d "%PROJECT_ROOT%*审查*" 2>nul
dir /b /s /a-d "%PROJECT_ROOT%*质检*" 2>nul

echo.
echo [10] 其他杂项:
dir /b /a-d "%PROJECT_ROOT%_p5_3*.json" 2>nul

echo.
echo ============================================
echo   预览完成
echo ============================================
echo.

echo.
```

### 3.2 执行模式

```batch
@echo off
setlocal EnableDelayedExpansion
chcp 65001 >nul

echo ============================================
echo   归档清理 - 执行模式
echo ============================================
echo.

set "PROJECT_ROOT=%~dp0"
set "ARCHIVE=%PROJECT_ROOT%_archive"

:: 创建归档目录结构
echo [INFO] 创建归档目录结构...
mkdir "%ARCHIVE%\01_临时脚本" 2>nul
mkdir "%ARCHIVE%\02_缓存数据" 2>nul
mkdir "%ARCHIVE%\02_缓存数据\data" 2>nul
mkdir "%ARCHIVE%\03_数据库备份" 2>nul
mkdir "%ARCHIVE%\04_PPT提取物" 2>nul
mkdir "%ARCHIVE%\05_过时文档" 2>nul
mkdir "%ARCHIVE%\05_过时文档\docs" 2>nul
mkdir "%ARCHIVE%\06_测试脚本" 2>nul
mkdir "%ARCHIVE%\07_工具脚本" 2>nul
mkdir "%ARCHIVE%\08_提示词备份" 2>nul
mkdir "%ARCHIVE%\09_审计报告" 2>nul
mkdir "%ARCHIVE%\10_其他杂项" 2>nul
echo [OK] 目录结构创建完成
echo.

set "MOVED=0"

:: 01_临时脚本
echo [01] 移动临时脚本...
for %%f in ("%PROJECT_ROOT%_*" "%PROJECT_ROOT%ocr_*.py" "%PROJECT_ROOT%ocr_windows.ps1") do (
    if exist "%%f" (
        move /y "%%f" "%ARCHIVE%\01_临时脚本\" >nul 2>&1
        if !errorlevel!==0 (
            set /a MOVED+=1
            echo      + %%~nxf
        )
    )
)
echo [OK] 01_临时脚本 完成
echo.

:: 02_缓存数据
echo [02] 移动缓存数据...
if exist "%PROJECT_ROOT%data\extracted" (
    move /y "%PROJECT_ROOT%data\extracted" "%ARCHIVE%\02_缓存数据\" >nul 2>&1
    if !errorlevel!==0 echo      + data\extracted/
)
if exist "%PROJECT_ROOT%data\ai_exports" (
    move /y "%PROJECT_ROOT%data\ai_exports" "%ARCHIVE%\02_缓存数据\" >nul 2>&1
    if !errorlevel!==0 echo      + data\ai_exports/
)
if exist "%PROJECT_ROOT%data\ai_imports" (
    move /y "%PROJECT_ROOT%data\ai_imports" "%ARCHIVE%\02_缓存数据\" >nul 2>&1
    if !errorlevel!==0 echo      + data\ai_imports/
)
echo [OK] 02_缓存数据 完成
echo.

:: 03_数据库备份
echo [03] 移动数据库备份...
for %%f in ("%PROJECT_ROOT%data\*backup*.db") do (
    if exist "%%f" (
        move /y "%%f" "%ARCHIVE%\03_数据库备份\" >nul 2>&1
        if !errorlevel!==0 (
            set /a MOVED+=1
            echo      + %%~nxf
        )
    )
)
echo [OK] 03_数据库备份 完成
echo.

:: 04_PPT提取物
echo [04] 移动PPT提取物...
if exist "%PROJECT_ROOT%src\notes\ppt_extracted" (
    move /y "%PROJECT_ROOT%src\notes\ppt_extracted" "%ARCHIVE%\04_PPT提取物\" >nul 2>&1
    if !errorlevel!==0 echo      + src\notes\ppt_extracted/
)
echo [OK] 04_PPT提取物 完成
echo.

:: 05_过时文档
echo [05] 移动过时文档...
for %%f in ("%PROJECT_ROOT%docs\*.md") do (
    echo "%%~nxf" | findstr /i "README.md 项目重启方案.md" >nul
    if errorlevel 1 (
        move /y "%%f" "%ARCHIVE%\05_过时文档\docs\" >nul 2>&1
        if !errorlevel!==0 echo      + %%~nxf
    )
)
:: 移动 docs/reports
if exist "%PROJECT_ROOT%docs\reports" (
    move /y "%PROJECT_ROOT%docs\reports" "%ARCHIVE%\05_过时文档\docs\" >nul 2>&1
    if !errorlevel!==0 echo      + docs\reports/
)
:: 移动 docs/项目文档
if exist "%PROJECT_ROOT%docs\项目文档" (
    move /y "%PROJECT_ROOT%docs\项目文档" "%ARCHIVE%\05_过时文档\docs\" >nul 2>&1
    if !errorlevel!==0 echo      + docs\项目文档/
)
echo [OK] 05_过时文档 完成
echo.

:: 06_测试脚本
echo [06] 移动测试脚本...
for %%f in ("%PROJECT_ROOT%scripts\p0_*" "%PROJECT_ROOT%scripts\p1_*" "%PROJECT_ROOT%scripts\debug_*" "%PROJECT_ROOT%scripts\fix_*" "%PROJECT_ROOT%scripts\check_*" "%PROJECT_ROOT%scripts\_crop_*" "%PROJECT_ROOT%scripts\_extract_*" "%PROJECT_ROOT%scripts\_fix_*" "%PROJECT_ROOT%scripts\_check_*" "%PROJECT_ROOT%scripts\link_*" "%PROJECT_ROOT%scripts\rebuild_*" "%PROJECT_ROOT%scripts\generate_*" "%PROJECT_ROOT%scripts\inspect_db*.py" "%PROJECT_ROOT%scripts\deep_verify.py" "%PROJECT_ROOT%scripts\scan_*") do (
    for %%g in ("%%f") do (
        if exist "%%g" (
            move /y "%%g" "%ARCHIVE%\06_测试脚本\" >nul 2>&1
```
